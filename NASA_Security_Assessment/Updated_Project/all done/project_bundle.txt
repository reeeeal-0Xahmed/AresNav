<?php

namespace Database\Seeders;

use App\Models\SciencePoi;
use Illuminate\Database\Seeder;
use Illuminate\Support\Facades\Http;

/**
 * Loads Perseverance (Mars 2020) drive waypoints from NASA JPL MMGIS.
 * NOTE: these are rover waypoints, not curated science targets. Add real science
 * targets (with their source) as extra rows of another `type` for the demo.
 * Run: php artisan db:seed --class=SciencePoiSeeder
 */
class SciencePoiSeeder extends Seeder
{
    private const SOURCE_URL = 'https://mars.nasa.gov/mmgis-maps/M20/Layers/json/M20_waypoints.json';
    private const KEEP_LAST = 60; // most recent waypoints that fall inside the map bounds

    public function run(): void
    {
        $bounds = config('mars.maps.jezero-regional-200m.bounds');
        $response = Http::timeout(15)->get(self::SOURCE_URL);
        if ($response->failed()) {
            $this->command->error('Could not reach NASA JPL MMGIS. Check the connection / URL.');
            return;
        }

        $inside = [];
        foreach ($response->json()['features'] ?? [] as $feature) {
            $c = $feature['geometry']['coordinates'] ?? [];
            if (count($c) < 2 || $c[0] < $bounds['west'] || $c[0] > $bounds['east'] || $c[1] < $bounds['south'] || $c[1] > $bounds['north']) {
                continue;
            }
            $inside[] = $feature;
        }
        $this->command->info(count($inside).' waypoints fall inside the map bounds.');

        $count = 0;
        foreach (array_slice($inside, -self::KEEP_LAST) as $feature) {
            $p = $feature['properties'] ?? [];
            $c = $feature['geometry']['coordinates'];
            $sol = $p['sol'] ?? $count;
            $site = $p['site'] ?? '0';
            $drive = $p['drive'] ?? '0';
            SciencePoi::updateOrCreate(
                ['poi_code' => "m20_sol_{$sol}_s{$site}_d{$drive}"],
                [
                    'name' => "Perseverance waypoint — site {$site}, drive {$drive} (sol {$sol})",
                    'description' => "Perseverance drive waypoint recorded on sol {$sol}.",
                    'lon' => $c[0],
                    'lat' => $c[1],
                    'type' => 'rover_waypoint',
                    'metadata' => [
                        'mission' => 'Mars 2020 Perseverance',
                        'sol' => $sol,
                        'elevation_m' => $c[2] ?? null,
                        'source' => 'NASA JPL MMGIS',
                    ],
                ]
            );
            $count++;
        }
        $this->command->info("Seeded {$count} waypoints.");
    }
}

<?php

namespace App\Services;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;
use Throwable;

/**
 * NASA DONKI solar-flare status used as a RADIATION PROXY.
 * DONKI reports flares as seen from Earth; this is NOT a Mars surface dose measurement.
 *
 * Fixes vs. the first version: failures return UNKNOWN (never "SAFE"), failures are
 * not cached, and the strongest flare in the window is used (not just the last item).
 */
final class SpaceWeather
{
    private const CACHE_KEY = 'nasa_donki_space_weather';
    private const WINDOW_DAYS = 7;
    private const LETTER_RANK = ['A' => 1, 'B' => 2, 'C' => 3, 'M' => 4, 'X' => 5];

    public function current(): array
    {
        $cached = Cache::get(self::CACHE_KEY);
        if (is_array($cached)) {
            return $cached;
        }

        try {
            $response = Http::timeout(10)->get('https://api.nasa.gov/DONKI/FLR', [
                'startDate' => now()->subDays(self::WINDOW_DAYS)->format('Y-m-d'),
                'endDate' => now()->format('Y-m-d'),
                'api_key' => config('services.nasa.key', 'DEMO_KEY'),
            ]);
            if ($response->successful() && is_array($response->json())) {
                $status = $this->classify($response->json());
                Cache::put(self::CACHE_KEY, $status, now()->addHours(3));
                return $status;
            }
            Log::warning('DONKI request failed', ['status' => $response->status()]);
        } catch (Throwable $e) {
            Log::warning('DONKI exception: '.$e->getMessage());
        }

        return $this->unknown();
    }

    /** @param list<array<string,mixed>> $flares */
    public function classify(array $flares): array
    {
        $strongest = null;
        $bestScore = -1.0;
        foreach ($flares as $flare) {
            $class = (string) ($flare['classType'] ?? '');
            if (!preg_match('/^([ABCMX])(\d+(?:\.\d+)?)$/', $class, $m)) {
                continue;
            }
            $score = self::LETTER_RANK[$m[1]] * 1000 + (float) $m[2];
            if ($score > $bestScore) {
                $bestScore = $score;
                $strongest = ['class' => $class, 'letter' => $m[1], 'mag' => (float) $m[2], 'time' => $flare['beginTime'] ?? null];
            }
        }

        $status = 'NOMINAL';
        if ($strongest !== null) {
            if ($strongest['letter'] === 'X') {
                $status = 'HIGH_RADIATION_WARNING';
            } elseif ($strongest['letter'] === 'M' && $strongest['mag'] >= 5.0) {
                $status = 'ELEVATED';
            }
        }

        return [
            'radiation_status' => $status,
            'strongest_flare_class' => $strongest['class'] ?? 'NONE',
            'event_time' => $strongest['time'] ?? null,
            'window_days' => self::WINDOW_DAYS,
            'basis' => 'earth_view_flare_proxy',
            'source' => 'NASA DONKI FLR',
            'checked_at' => now()->toIso8601String(),
        ];
    }

    private function unknown(): array
    {
        return [
            'radiation_status' => 'UNKNOWN',
            'strongest_flare_class' => null,
            'event_time' => null,
            'window_days' => self::WINDOW_DAYS,
            'basis' => 'earth_view_flare_proxy',
            'source' => 'NASA DONKI FLR (unavailable)',
            'checked_at' => now()->toIso8601String(),
        ];
    }
}

<?php

namespace App\Services;

use App\Exceptions\MarsPlanningException;
use Illuminate\Contracts\Config\Repository;
use JsonException;

final class MarsTerrainMap
{
    public function __construct(private readonly Repository $config) {}

    public function load(string $mapId): array
    {
        $settings = $this->config->get('mars.maps.'.$mapId);
        if (!is_array($settings)) {
            throw new MarsPlanningException('الخريطة المطلوبة غير موجودة.', 'MAP_NOT_FOUND', 404);
        }
        $file = $settings['terrain_file'] ?? '';
        if (!is_string($file) || !is_file($file) || !is_readable($file)) {
            throw new MarsPlanningException('ملف التضاريس غير متاح على السيرفر.', 'TERRAIN_UNAVAILABLE');
        }
        try {
            $text = file_get_contents($file);
            if ($text === false) {
                throw new MarsPlanningException('تعذر قراءة التضاريس.', 'TERRAIN_UNAVAILABLE');
            }
            $data = json_decode($text, true, 512, JSON_THROW_ON_ERROR);
        } catch (JsonException $exception) {
            throw new MarsPlanningException('ملف التضاريس ليس JSON صالحًا.', 'INVALID_TERRAIN');
        }
        if (!is_array($data)) {
            throw new MarsPlanningException('ملف التضاريس غير صالح.', 'INVALID_TERRAIN');
        }

        $width = $data['width'] ?? null;
        $height = $data['height'] ?? null;
        $grid = $data['grid'] ?? null;
        $metadata = $data['terrainMetadata'] ?? null;
        if (!is_int($width) || !is_int($height) || $width <= 0 || $height <= 0 || $width * $height > 1_000_000 ||
            !is_array($grid) || !array_is_list($grid) || count($grid) !== $height || !is_array($metadata)) {
            throw new MarsPlanningException('أبعاد أو بيانات التضاريس غير صالحة.', 'INVALID_TERRAIN');
        }
        foreach ($grid as $row) {
            if (!is_array($row) || !array_is_list($row) || count($row) !== $width) {
                throw new MarsPlanningException('صفوف التضاريس غير متطابقة.', 'INVALID_TERRAIN');
            }
            foreach ($row as $cost) {
                if (!is_int($cost) || !in_array($cost, [-1, 1, 2, 4, 8], true)) {
                    throw new MarsPlanningException('تكلفة الخلية غير صالحة.', 'INVALID_TERRAIN');
                }
            }
        }

        MarsCoordinates::assertMetadata($metadata);
        $cellSize = (float) $metadata['cellSizeMeters'];
        try {
            $version = hash('sha256', json_encode([
                'width' => $width,
                'height' => $height,
                'grid' => $grid,
                'terrainMetadata' => $metadata,
            ], JSON_THROW_ON_ERROR));
        } catch (JsonException $exception) {
            throw new MarsPlanningException('تعذر تثبيت نسخة التضاريس.', 'INVALID_TERRAIN');
        }

        return [
            'map_id' => $mapId,
            'title' => $settings['title'] ?? $mapId,
            'version' => $version,
            'width' => $width,
            'height' => $height,
            'cell_size_meters' => $cellSize,
            'grid' => $grid,
            'terrain_metadata' => $metadata,
            'projection' => $metadata['projection'],
            'affine' => MarsCoordinates::assertAffine($metadata['affine']),
            'requested_bounds' => $metadata['requestedBounds'],
            'raster_bounds' => $metadata['rasterBounds'],
            'regional_scale' => $metadata['navigationScale'] ?? 'regional',
            'warnings' => $metadata['warnings'],
            'thresholds' => $metadata['thresholds'],
            'source' => $this->safeSource($metadata),
        ];
    }

    public function publicPayload(array $map): array
    {
        $metadata = $map['terrain_metadata'];
        $payload = [
            'map_id' => $map['map_id'],
            'title' => $map['title'],
            'width' => $map['width'],
            'height' => $map['height'],
            'grid' => $map['grid'],
            'version' => $map['version'],
            'projection' => $map['projection'],
            'affine' => $map['affine'],
            'requested_bounds' => $map['requested_bounds'],
            'raster_bounds' => $map['raster_bounds'],
            'regional_scale' => $map['regional_scale'],
            'warnings' => $map['warnings'],
            'source' => $map['source'],
            'thresholds' => $map['thresholds'],
            'legend' => [
                'costs' => [-1, 1, 2, 4, 8],
                'labels' => ['-1' => 'blocked', '1' => 'low', '2' => 'moderate', '4' => 'high', '8' => 'very_high'],
            ],
            'cell_size_meters' => $map['cell_size_meters'],
        ];
        return $payload;
    }

    public function resultMetadata(array $map): array
    {
        $metadata = $map['terrain_metadata'];
        unset($metadata['sourceDem']);
        return $metadata;
    }

    public function coordinateToCell(array $coordinate, array $map, string $label): array
    {
        MarsCoordinates::assertFiniteNumber($coordinate['lon'] ?? null, $label.' longitude');
        MarsCoordinates::assertFiniteNumber($coordinate['lat'] ?? null, $label.' latitude');
        $bounds = $map['requested_bounds'];
        if ((float) $coordinate['lon'] < (float) $bounds['west'] || (float) $coordinate['lon'] > (float) $bounds['east'] ||
            (float) $coordinate['lat'] < (float) $bounds['south'] || (float) $coordinate['lat'] > (float) $bounds['north']) {
            throw new MarsPlanningException($label.' خارج الحدود الجغرافية المطلوبة.', 'OUT_OF_BOUNDS', 422);
        }
        $projected = MarsCoordinates::project($coordinate, $map['terrain_metadata']);
        $affine = $map['affine'];
        $determinant = $affine[0] * $affine[4] - $affine[1] * $affine[3];
        if (abs($determinant) < 1e-12) {
            throw new MarsPlanningException('تحويل التضاريس غير قابل للعكس.', 'INVALID_TERRAIN');
        }
        $dx = $projected['x'] - $affine[2];
        $dy = $projected['y'] - $affine[5];
        $column = ($affine[4] * $dx - $affine[1] * $dy) / $determinant;
        $row = (-$affine[3] * $dx + $affine[0] * $dy) / $determinant;
        return ['x' => (int) floor($column), 'y' => (int) floor($row)];
    }

    public function cellCenter(array $map, array $point): array
    {
        return MarsCoordinates::cellCenter((int) $point['x'], (int) $point['y'], $map['terrain_metadata']);
    }

    public function assertUsablePoint(array $map, array $point, string $label): void
    {
        $x = $point['x'] ?? null;
        $y = $point['y'] ?? null;
        if (!is_int($x) || !is_int($y) || $x < 0 || $x >= $map['width'] || $y < 0 || $y >= $map['height']) {
            throw new MarsPlanningException($label.' خارج حدود الشبكة.', 'OUT_OF_BOUNDS', 422);
        }
        if ($map['grid'][$y][$x] < 0) {
            throw new MarsPlanningException($label.' تقع على خلية ممنوعة.', 'BLOCKED_POINT', 422);
        }
    }

    private function safeSource(array $metadata): array
    {
        $source = is_array($metadata['source'] ?? null) ? $metadata['source'] : [];
        $sourceUrl = $metadata['sourceUrl'] ?? null;
        if (is_string($sourceUrl) && !isset($source['url']) && !isset($source['originalUrl'])) {
            $source['url'] = $sourceUrl;
        }
        unset($source['sourceDem'], $source['path'], $source['file'], $source['filename']);
        $source['nominal_cell_size_meters'] = $source['nominalPixelSpacingMeters'] ?? $metadata['cellSizeMeters'] ?? null;
        $source['navigation_scale'] = $metadata['navigationScale'] ?? 'regional';
        return $source;
    }
}

<?php

namespace App\Services;

use App\Exceptions\MarsPlanningException;

final class MarsCoordinates
{
    public const RADIUS_METERS = 3396190.0;

    /** @param array{lon:int|float,lat:int|float} $coordinate */
    public static function project(array $coordinate, array $metadata): array
    {
        self::assertFiniteNumber($coordinate['lon'] ?? null, 'longitude');
        self::assertFiniteNumber($coordinate['lat'] ?? null, 'latitude');
        self::assertProjectionMetadata($metadata);

        $projection = $metadata['projection'];
        $latTs = self::radians((float) $projection['standardParallelDegrees']);
        $lon0 = (float) $projection['centralLongitudeDegrees'];
        $lat0 = (float) $projection['latitudeOriginDegrees'];
        $radius = (float) $projection['radiusMeters'];

        return [
            'x' => $radius * cos($latTs) * self::radians((float) $coordinate['lon'] - $lon0),
            'y' => $radius * self::radians((float) $coordinate['lat'] - $lat0),
        ];
    }

    /** @return array{lon:float,lat:float} */
    public static function inverse(array $projected, array $metadata): array
    {
        self::assertFiniteNumber($projected['x'] ?? null, 'projected x');
        self::assertFiniteNumber($projected['y'] ?? null, 'projected y');
        self::assertProjectionMetadata($metadata);

        $projection = $metadata['projection'];
        $latTs = self::radians((float) $projection['standardParallelDegrees']);
        $lon0 = (float) $projection['centralLongitudeDegrees'];
        $lat0 = (float) $projection['latitudeOriginDegrees'];
        $radius = (float) $projection['radiusMeters'];

        return [
            'lon' => rad2deg((float) $projected['x'] / ($radius * cos($latTs))) + $lon0,
            'lat' => rad2deg((float) $projected['y'] / $radius) + $lat0,
        ];
    }

    /** @return array{lon:float,lat:float} */
    public static function cellCenter(int $x, int $y, array $metadata): array
    {
        $affine = self::assertAffine($metadata['affine'] ?? null);
        $projected = [
            'x' => $affine[0] * ($x + 0.5) + $affine[1] * ($y + 0.5) + $affine[2],
            'y' => $affine[3] * ($x + 0.5) + $affine[4] * ($y + 0.5) + $affine[5],
        ];
        return self::inverse($projected, $metadata);
    }

<?php

namespace App\Services;

use RuntimeException;

/**
 * Turns a planned route into a mission briefing: terrain stats, EVA time,
 * nearby science points and an explainable GO / CAUTION / NO-GO verdict.
 *
 * Framework-free on purpose (arrays in, arrays out) so it can be unit-tested
 * without booting Laravel. Every threshold is a PLANNING ASSUMPTION, not a
 * NASA operational limit; they are overridable through config('mars.briefing').
 */
final class RouteBriefing
{
    private const MARS_RADIUS_M = 3396190.0;

    private const DEFAULTS = [
        'walk_speed_kmh' => 2.0,          // assumed suited walking speed
        'eva_budget_hours' => 8.0,        // assumed EVA duration budget
        'eva_tight_share' => 0.7,         // round trip above this share of budget => CAUTION
        'poi_radius_m' => 2000.0,         // science points within this distance of the path
        'slope_caution_deg' => 10.0,
        'slope_nogo_deg' => 20.0,
        'high_cost_share_caution' => 0.25, // share of cost>=4 cells => CAUTION
        'max_pois' => 10,
    ];

    private array $opt;

    /** @param list<list<float>> $elevations [y][x] metres, same grid as terrain.json */
    public function __construct(private readonly array $elevations, array $options = [])
    {
        $this->opt = array_replace(self::DEFAULTS, $options);
    }

    public static function fromFile(string $file, array $options = []): self
    {
        $text = is_file($file) && is_readable($file) ? file_get_contents($file) : false;
        $data = $text === false ? null : json_decode($text, true);
        if (!is_array($data) || !isset($data['elevations']) || !is_array($data['elevations'])) {
            throw new RuntimeException('Elevation file missing or invalid: '.$file);
        }
        return new self($data['elevations'], $options);
    }

    /**
     * Adds ['briefing'] to every path (and 'verdict' to its GeoJSON properties).
     * Only touches successful results; returns everything else unchanged.
     *
     * @param list<array{lon:float,lat:float,id?:string,name?:string,type?:string}> $pois
     */
    public function attach(array $result, array $grid, float $cellSize, array $pois, ?array $radiation): array
    {
        if (($result['status'] ?? '') !== 'success' || empty($result['paths'])) {
            return $result;
        }
        foreach ($result['paths'] as $i => $route) {
            $briefing = $this->build(
                $route['path'] ?? [],
                $route['geographic_path'] ?? [],
                $grid,
                $cellSize,
                (float) ($route['distance_meters'] ?? 0),
                $pois,
                $radiation,
            );
            $result['paths'][$i]['briefing'] = $briefing;
            if (isset($result['paths'][$i]['geojson']['properties'])) {
                $result['paths'][$i]['geojson']['properties']['verdict'] = $briefing['verdict'];
            }
            if (isset($result['geojson']['features'][$i]['properties'])) {
                $result['geojson']['features'][$i]['properties']['verdict'] = $briefing['verdict'];
            }
        }
        $result['verdict'] = $result['paths'][0]['briefing']['verdict'];
        return $result;
    }

    public function build(
        array $path,
        array $geoPath,
        array $grid,
        float $cellSize,
        float $distanceMeters,
        array $pois,
        ?array $radiation,
    ): array {
        $slopes = [];
        $ascent = 0.0;
        $descent = 0.0;
        $classes = [1 => 0, 2 => 0, 4 => 0, 8 => 0];
        $minElev = INF;
        $maxElev = -INF;

        foreach ($path as $i => $p) {
            $z = (float) ($this->elevations[$p['y']][$p['x']] ?? NAN);
            if (is_finite($z)) {
                $minElev = min($minElev, $z);
                $maxElev = max($maxElev, $z);
            }
            if ($i === 0) {
                continue; // entering a cell pays its cost; the start cell is free
            }
            $cost = $grid[$p['y']][$p['x']] ?? null;
            if (isset($classes[$cost])) {
                $classes[$cost]++;
            }
            $prev = $path[$i - 1];
            $zPrev = (float) ($this->elevations[$prev['y']][$prev['x']] ?? NAN);
            if (!is_finite($z) || !is_finite($zPrev)) {
                continue;
            }
            $dz = $z - $zPrev;
            $run = $cellSize * hypot($p['x'] - $prev['x'], $p['y'] - $prev['y']);
            $slopes[] = rad2deg(atan(abs($dz) / max($run, 1e-9)));
            $dz > 0 ? $ascent += $dz : $descent -= $dz;
        }

        $steps = max(count($path) - 1, 0);
        sort($slopes);
        $p95 = $slopes === [] ? 0.0 : $slopes[(int) max(ceil(0.95 * count($slopes)) - 1, 0)];
        $highShare = $steps > 0 ? ($classes[4] + $classes[8]) / $steps : 0.0;

        $oneWayHours = ($distanceMeters / 1000.0) / $this->opt['walk_speed_kmh'];
        $roundTripHours = 2.0 * $oneWayHours;
        $stops = $this->sciencePoints($geoPath, $pois);
        $radiationStatus = (string) ($radiation['radiation_status'] ?? 'UNKNOWN');

        [$verdict, $reasons] = $this->verdict(
            $roundTripHours,
            $slopes === [] ? 0.0 : max($slopes),
            $highShare,
            $radiationStatus,
        );

        return [
            'verdict' => $verdict,
            'reasons' => $reasons,
            'terrain' => [
                'distance_m' => $distanceMeters,
                'steps' => $steps,
                'max_slope_deg' => round($slopes === [] ? 0.0 : max($slopes), 2),
                'mean_slope_deg' => round($slopes === [] ? 0.0 : array_sum($slopes) / count($slopes), 2),
                'p95_slope_deg' => round($p95, 2),
                'ascent_m' => round($ascent, 1),
                'descent_m' => round($descent, 1),
                'elevation_min_m' => is_finite($minElev) ? round($minElev, 1) : null,
                'elevation_max_m' => is_finite($maxElev) ? round($maxElev, 1) : null,
                'cost_cells' => $classes,
                'high_cost_share' => round($highShare, 3),
            ],
            'eva' => [
                'one_way_hours' => round($oneWayHours, 2),
                'round_trip_hours' => round($roundTripHours, 2),
                'budget_hours' => $this->opt['eva_budget_hours'],
                'walk_speed_kmh' => $this->opt['walk_speed_kmh'],
                'basis' => 'planning_assumption',
            ],
            'radiation' => [
                'status' => $radiationStatus,
                'basis' => $radiation['basis'] ?? null,
            ],
            'science_stops' => $stops,
            'limitations' => [
                'Slope is measured between 200 m cell centres; local slopes are steeper.',
                'Radiation status is an Earth-view solar-flare proxy, not a Mars surface measurement.',
                'Speeds, EVA budget and thresholds are planning assumptions, not NASA operational limits.',
            ],
        ];
    }

    /** @return array{0:string,1:list<array{code:string,severity:string,detail:string}>} */
    private function verdict(float $roundTripHours, float $maxSlope, float $highShare, string $radiation): array
    {
        $level = 0;
        $reasons = [];
        $add = function (int $sev, string $code, string $detail) use (&$level, &$reasons): void {
            $level = max($level, $sev);
            $reasons[] = ['code' => $code, 'severity' => $sev === 2 ? 'NO-GO' : 'CAUTION', 'detail' => $detail];
        };

        $budget = (float) $this->opt['eva_budget_hours'];
        if ($roundTripHours > $budget) {
            $add(2, 'EVA_TIME_EXCEEDED', sprintf('Round trip ~%.1f h exceeds the %.1f h EVA budget.', $roundTripHours, $budget));
        } elseif ($roundTripHours > $budget * $this->opt['eva_tight_share']) {
            $add(1, 'EVA_TIME_TIGHT', sprintf('Round trip ~%.1f h uses most of the %.1f h EVA budget.', $roundTripHours, $budget));
        }
        if ($maxSlope >= $this->opt['slope_nogo_deg']) {
            $add(2, 'SLOPE_LIMIT', sprintf('Max slope %.1f° reaches the %.0f° limit.', $maxSlope, $this->opt['slope_nogo_deg']));
        } elseif ($maxSlope >= $this->opt['slope_caution_deg']) {
            $add(1, 'STEEP_SECTION', sprintf('Max slope %.1f° on at least one step.', $maxSlope));
        }
        if ($highShare > $this->opt['high_cost_share_caution']) {
            $add(1, 'HIGH_COST_TERRAIN', sprintf('%.0f%% of the route crosses high-cost terrain.', $highShare * 100));
        }
        match ($radiation) {
            'HIGH_RADIATION_WARNING' => $add(2, 'RADIATION_HIGH', 'Strong recent solar flare activity.'),
            'ELEVATED' => $add(1, 'RADIATION_ELEVATED', 'Moderate recent solar flare activity.'),
            'NOMINAL' => null,
            default => $add(1, 'RADIATION_UNKNOWN', 'Space-weather data unavailable; treat as unverified.'),
        };

        return [['GO', 'CAUTION', 'NO-GO'][$level], $reasons];
    }

    /** @return list<array<string,mixed>> */
    private function sciencePoints(array $geoPath, array $pois): array
    {
        if ($geoPath === []) {
            return [];
        }
        $out = [];
        foreach ($pois as $poi) {
            if (!isset($poi['lon'], $poi['lat'])) {
                continue;
            }
            $cosLat = cos(deg2rad((float) $poi['lat']));
            $best = INF;
            $bestIdx = 0;
            foreach ($geoPath as $i => $g) {
                $dx = deg2rad($g['lon'] - $poi['lon']) * $cosLat * self::MARS_RADIUS_M;
                $dy = deg2rad($g['lat'] - $poi['lat']) * self::MARS_RADIUS_M;
                $d2 = $dx * $dx + $dy * $dy;
                if ($d2 < $best) {
                    $best = $d2;
                    $bestIdx = $i;
                }
            }
            $dist = sqrt($best);
            if ($dist <= $this->opt['poi_radius_m']) {
                $out[] = [
                    'id' => $poi['id'] ?? null,
                    'name' => $poi['name'] ?? null,
                    'type' => $poi['type'] ?? null,
                    'distance_to_path_m' => round($dist),
                    'nearest_path_index' => $bestIdx,
                ];
            }
        }
        usort($out, static fn (array $a, array $b): int => $a['distance_to_path_m'] <=> $b['distance_to_path_m']);
        return array_slice($out, 0, (int) $this->opt['max_pois']);
    }
}

                $x = $point['x'] ?? null;
                $y = $point['y'] ?? null;
                if (!is_int($x) || !is_int($y) || $x < 0 || $x >= $map['width'] || $y < 0 || $y >= $map['height'] || $map['grid'][$y][$x] < 0) {
                    throw new MarsPlanningException('المسار يدخل خلية غير صالحة.', 'INVALID_PLANNER_OUTPUT', 502);
                }
                $id = $y * $map['width'] + $x;
                if (isset($seen[$id])) {
                    throw new MarsPlanningException('المسار يكرر خلية.', 'INVALID_PLANNER_OUTPUT', 502);
                }
                $seen[$id] = true;
                if ($i > 0) {
                    $before = $points[$i - 1];
                    if (abs($before['x'] - $x) + abs($before['y'] - $y) !== 1) {
                        throw new MarsPlanningException('المسار يحتوي قفزة غير مسموحة.', 'INVALID_PLANNER_OUTPUT', 502);
                    }
                    $calculatedCost += $map['grid'][$y][$x];
                }
            }
            if (abs($calculatedCost - $cost) > 1e-8 * max(1.0, abs($calculatedCost))) {
                throw new MarsPlanningException('تكلفة المسار لا تطابق الشبكة.', 'INVALID_PLANNER_OUTPUT', 502);
            }
            $hash = hash('sha256', json_encode($points, JSON_THROW_ON_ERROR));
            if (isset($seenRoutes[$hash])) {
                throw new MarsPlanningException('برنامج المسارات كرر نفس المسار.', 'INVALID_PLANNER_OUTPUT', 502);
            }
            $seenRoutes[$hash] = true;
            $route['rank'] = $index + 1;
            $route['steps'] = count($points) - 1;
            $route['distance_meters'] = $route['steps'] * $map['cell_size_meters'];
            $previousCost = $cost;
        }
        unset($route);
        $result['paths'] = $paths;
        $result['path'] = $paths[0]['path'];
        return $result;
    }
}

    /**

    private function withBriefing(array $result, array $map, SpaceWeather $weather): array
    {
        try {

    /** @return array{kind:string,requested:array,cell:array{x:int,y:int}} */
    private function normalizeInputCoordinate(mixed $value, array $map, MarsTerrainMap $terrain, string $field): array
    {
        if (!is_array($value)) {
            $this->invalidCoordinate($field, 'الإحداثيات يجب أن تكون كائنًا.');
        }
        $keys = array_keys($value);
        sort($keys);
        if ($keys === ['x', 'y']) {
            if (!is_int($value['x'] ?? null) || !is_int($value['y'] ?? null)) {
                $this->invalidCoordinate($field, 'إحداثيات الشبكة يجب أن تكون أعدادًا صحيحة.');
            }
            $cell = ['x' => $value['x'], 'y' => $value['y']];
            if ($cell['x'] < 0 || $cell['x'] >= $map['width'] || $cell['y'] < 0 || $cell['y'] >= $map['height']) {
                $this->invalidCoordinate($field, 'إحداثيات الشبكة خارج حدود الشبكة.');
            }
            return ['kind' => 'grid', 'requested' => $cell, 'cell' => $cell];
        }
        if ($keys === ['lat', 'lon']) {
            foreach (['lon', 'lat'] as $coordinate) {
                if (!(is_int($value[$coordinate] ?? null) || is_float($value[$coordinate] ?? null)) || !is_finite((float) $value[$coordinate])) {
                    $this->invalidCoordinate($field, 'الإحداثيات الجغرافية يجب أن تكون أعدادًا حقيقية منتهية.');
                }
            }
            $requested = ['lon' => $value['lon'], 'lat' => $value['lat']];
            $bounds = $map['requested_bounds'];
            if ($requested['lon'] < $bounds['west'] || $requested['lon'] > $bounds['east'] ||
                $requested['lat'] < $bounds['south'] || $requested['lat'] > $bounds['north']) {
                $this->invalidCoordinate($field, 'الإحداثيات الجغرافية خارج الحدود المطلوبة.');
            }
            return ['kind' => 'geographic', 'requested' => $requested, 'cell' => $terrain->coordinateToCell($requested, $map, $field)];
        }
        $this->invalidCoordinate($field, 'استخدم بالضبط x,y أو بالضبط lon,lat دون خلط.');
    }

    private function invalidCoordinate(string $field, string $message): never
    {
        throw ValidationException::withMessages([$field => [$message]]);
    }

SOURCE_URL = 'https://asc-pds-services.s3.us-west-2.amazonaws.com/mosaic/Mars/HRSC_MOLA_Blend/Mars_HRSC_MOLA_BlendDEM_Global_200mp_v2.tif'

DEFAULT_TREK_URL = 'https://trek.nasa.gov/mars/#v=0.1&x=77.42982155&y=18.46611095&z=9&p=urn%3Aogc%3Adef%3Acrs%3AEPSG%3A%3A104905&d=&locale=&b=mars&e=76.80291421306124%2C17.96348644765699%2C78.05672888693876%2C18.96873545234301&sfz=&w='
RADIUS = 3396190.0


def log(text):
    print(text, file=sys.stderr, flush=True)


def parse_extent(url):
    parsed = urlparse(url)
    if parsed.hostname != 'trek.nasa.gov' or not parsed.path.startswith('/mars/'):
        raise ValueError('الرابط يجب أن يكون Mars Trek الرسمي.')
    params = parse_qs(parsed.fragment)
    try:
        west, south, east, north = map(float, params['e'][0].split(','))
    except (KeyError, ValueError, IndexError):
        raise ValueError('الرابط لا يحتوي حدود e صالحة.')
    if not all(map(math.isfinite, [west, south, east, north])) or not (-180 <= west < east <= 180 and -90 < south < north < 90):
        raise ValueError('نطاق غير صالح أو يعبر خط تغيير longitude؛ هذه النسخة لا تدعم عبوره.')
    return {'west': west, 'south': south, 'east': east, 'north': north}


def bands(array, limits):
    out = np.ones(array.shape, dtype=np.int16)
    for threshold, cost in zip(limits, (2, 4, 8)):
        out[array >= threshold] = cost
    return out


def run(args):
    if not math.isfinite(args.cell_m) or args.cell_m < 200:
        raise ValueError('المصدر اسميًا 200 م؛ هذه النسخة تمنع إخراج شبكة أدق زائفة.')
    if not (10 < args.block_slope_deg < 90) or not math.isfinite(args.block_localdev_m) or args.block_localdev_m <= 60:
        raise ValueError('حدود النموذج يجب أن تتجاوز آخر درجات التكلفة: الميل >10، والانحراف المحلي >60 م.')
    extent = parse_extent(args.trek_url)
    center_lon = (extent['west'] + extent['east']) / 2
    center_lat = (extent['south'] + extent['north']) / 2
    geographic = CRS.from_proj4(f'+proj=longlat +R={RADIUS} +no_defs')
    metric = CRS.from_proj4(f'+proj=eqc +lat_ts={center_lat} +lon_0={center_lon} +lat_0={center_lat} +R={RADIUS} +units=m +no_defs')
    project = Transformer.from_crs(geographic, metric, always_xy=True)
    inverse = Transformer.from_crs(metric, geographic, always_xy=True)
    xmin, ymin = project.transform(extent['west'], extent['south'])
    xmax, ymax = project.transform(extent['east'], extent['north'])
    width, height = math.ceil((xmax - xmin) / args.cell_m), math.ceil((ymax - ymin) / args.cell_m)
    if width * height > 1_000_000:
        raise ValueError('شبكة أكبر من مليون خلية؛ زِد حجم خلية التخطيط بدل رفع الحد بلا تصميم جديد.')
    # Symmetric sub-cell padding: all cell centers remain inside requested bounds.
    xpad = (width * args.cell_m - (xmax - xmin)) / 2
    ypad = (height * args.cell_m - (ymax - ymin)) / 2
    left, top = xmin - xpad, ymax + ypad
    transform = from_origin(left, top, args.cell_m, args.cell_m)
    halo = 4  # Mean radius2 + slope radius1 + blocked buffer1; no artificial map border.
    halo_transform = from_origin(left - halo * args.cell_m, top + halo * args.cell_m, args.cell_m, args.cell_m)
    hw, hh = width + 2 * halo, height + 2 * halo
    halo_bounds = (halo_transform.c, halo_transform.f - hh * args.cell_m,
                   halo_transform.c + hw * args.cell_m, halo_transform.f)
    args.out.mkdir(parents=True, exist_ok=True)
    cache = args.out / 'native_source_window.tif'
    cache_metadata_file = args.out / 'native_source_window_metadata.json'
    cache_key = {'source': args.source, 'extent': extent, 'cellSizeMeters': args.cell_m}
    cache_metadata = json.loads(cache_metadata_file.read_text()) if cache_metadata_file.exists() else None
    use_cache = (cache.exists() and not args.refresh and isinstance(cache_metadata, dict)
                 and cache_metadata.get('key') == cache_key)
    source = str(cache) if use_cache else args.source
    log(f'النطاق كاملًا: {width} × {height}، خلية {args.cell_m} م؛ مساحة نحو {(xmax-xmin)/1000:.2f} × {(ymax-ymin)/1000:.2f} كم.')
    env = dict(GDAL_DISABLE_READDIR_ON_OPEN='EMPTY_DIR', CPL_VSIL_CURL_ALLOWED_EXTENSIONS='.tif,.tiff',
               GDAL_HTTP_TIMEOUT='90', GDAL_HTTP_CONNECTTIMEOUT='30', GDAL_HTTP_MAX_RETRY='3',
               GDAL_HTTP_RETRY_DELAY='2', GDAL_CACHEMAX=128,
               CPL_VSIL_CURL_CHUNK_SIZE='1048576', CPL_VSIL_CURL_CACHE_SIZE='134217728')
    with rasterio.Env(**env):
        with rasterio.open(source) as src:
            if src.count != 1 or src.crs is None:
                raise ValueError('المصدر يجب أن يكون DEM أحادي الطبقة وله مرجع إحداثيات.')
            radius = CRS.from_wkt(src.crs.to_wkt()).ellipsoid.semi_major_metre
            if not math.isclose(radius, RADIUS, abs_tol=1):
                raise ValueError('المصدر ليس مرجع المريخ؛ لن نستخدم EPSG:4326 الخاص بالأرض.')
            sb = transform_bounds(metric.to_wkt(), src.crs, *halo_bounds, densify_pts=21)
            floating = from_bounds(*sb, transform=src.transform)
            x0, y0 = max(0, math.floor(floating.col_off)-3), max(0, math.floor(floating.row_off)-3)
            x1 = min(src.width, math.ceil(floating.col_off+floating.width)+3)
            y1 = min(src.height, math.ceil(floating.row_off+floating.height)+3)
            if x1 <= x0 or y1 <= y0:
                raise ValueError('النطاق خارج المصدر.')
            values = np.full((y1-y0, x1-x0), np.nan, dtype=np.float32)
            log(f'قراءة نافذة الارتفاع {x1-x0} × {y1-y0} فقط. قد تُنقل شرائح أكبر من القص بسبب تخطيط GeoTIFF.')
            for row in range(y0, y1, 16):
                count = min(16, y1-row)
                band = src.read(1, window=Window(x0, row, x1-x0, count), masked=True)
                block = band.astype(np.float32).filled(np.nan) * float(src.scales[0]) + float(src.offsets[0])
                values[row-y0:row-y0+count] = block
                log(f'قراءة المصدر: {row-y0+count}/{y1-y0} صفًا.')
            source_transform = src.window_transform(Window(x0, y0, x1-x0, y1-y0))
            source_crs = src.crs.to_wkt()

Point = Tuple[int, int]
Route = Tuple[float, Tuple[Point, ...]]


def neighbors(point: Point, width: int, height: int) -> Iterable[Point]:
    x, y = point
    # This order is independent of solver tie ordering; tests compare equal
    # cost route identities as a set while requiring the sorted cost prefix.
    for dx, dy in ((0, -1), (0, 1), (-1, 0), (1, 0)):
        nx, ny = x + dx, y + dy
        if 0 <= nx < width and 0 <= ny < height:
            yield nx, ny


def enumerate_simple_routes(
    grid: Sequence[Sequence[float]], start: Point, target: Point
) -> List[Route]:
    """Enumerate all simple routes, paying the value of each entered cell."""
    height = len(grid)
    width = len(grid[0])
    routes: List[Route] = []

    def visit(point: Point, path: List[Point], cost: float) -> None:
        if point == target:
            routes.append((cost, tuple(path)))
            return
        for nxt in neighbors(point, width, height):
            if nxt in path or grid[nxt[1]][nxt[0]] < 0:
                continue
            visit(nxt, path + [nxt], cost + float(grid[nxt[1]][nxt[0]]))

    if grid[start[1]][start[0]] >= 0 and grid[target[1]][target[0]] >= 0:
        visit(start, [start], 0.0)
    return sorted(routes, key=lambda item: (item[0], item[1]))


def make_case(grid: Sequence[Sequence[float]], start: Point, target: Point) -> Dict:
    return {
        "width": len(grid[0]),
        "height": len(grid),
        "start": {"x": start[0], "y": start[1]},
        "target": {"x": target[0], "y": target[1]},
        "grid": [list(row) for row in grid],
    }

struct QueueItem {
  double priority;
  double pathCost;
  int node;
};

struct QueueItemGreater {
  bool operator()(const QueueItem &a, const QueueItem &b) const {
    if (a.priority != b.priority) return a.priority > b.priority;
    if (a.pathCost != b.pathCost) return a.pathCost < b.pathCost;
    return a.node > b.node;
  }
};

struct RouteGreater {
  bool operator()(const Route &a, const Route &b) const {
    if (a.cost != b.cost) return a.cost > b.cost;
    return a.nodes > b.nodes;
  }
};

// Reverse Dijkstra on the intact grid. For a forward edge predecessor ->
// current, the edge cost is grid[current], so reverse relaxation pays the
// current node's value rather than the predecessor's value. Negative cells
// are obstacles and are excluded from the reverse graph.
std::vector<double> reverseDistancesToTarget(
    const std::vector<std::vector<double>> &grid, int width, int height,
    int target) {
  const int nodeCount = width * height;
  const double infinity = std::numeric_limits<double>::infinity();
  std::vector<double> distance(nodeCount, infinity);
  std::priority_queue<QueueItem, std::vector<QueueItem>, QueueItemGreater>
      open;

  distance[target] = 0.0;
  open.push({0.0, 0.0, target});

  const int dx[4] = {0, 0, -1, 1};
  const int dy[4] = {-1, 1, 0, 0};

  while (!open.empty()) {
    const QueueItem current = open.top();
    open.pop();
    if (current.pathCost != distance[current.node]) continue;

    const auto [x, y] = fromNodeId(current.node, width);
    if (grid[y][x] < 0.0) continue;
    for (int direction = 0; direction < 4; ++direction) {
      const int px = x + dx[direction];
      const int py = y + dy[direction];
      if (px < 0 || px >= width || py < 0 || py >= height) continue;

      const int predecessor = toNodeId(px, py, width);
      if (grid[py][px] < 0.0) continue;
      const double predecessorDistance = current.pathCost + grid[y][x];
      if (predecessorDistance < distance[predecessor]) {
        distance[predecessor] = predecessorDistance;
        open.push({predecessorDistance, predecessorDistance, predecessor});
      }
    }
  }
  return distance;
}

// A* on the grid after removing banned nodes/edges. The heuristic is the
// original-grid distance to target, so bans can only make it more optimistic.
// The cost of the starting cell is excluded; each move pays the cost of the
// cell entered. Every negative grid value is treated as an obstacle.
bool shortestPath(const std::vector<std::vector<double>> &grid, int width,
                  int height, int start, int target,
                  const std::set<int> &bannedNodes,
                  const std::set<std::pair<int, int>> &bannedEdges,
                  const std::vector<double> &heuristic,
                  Route &result) {
  const int nodeCount = width * height;
  if (bannedNodes.count(start) || bannedNodes.count(target)) return false;

  std::vector<double> distance(nodeCount, std::numeric_limits<double>::infinity());
  std::vector<int> parent(nodeCount, -1);
  std::priority_queue<QueueItem, std::vector<QueueItem>, QueueItemGreater> open;

  distance[start] = 0.0;
  open.push({heuristic[start], 0.0, start});

  const int dx[4] = {0, 0, -1, 1};
  const int dy[4] = {-1, 1, 0, 0};

  while (!open.empty()) {
    const QueueItem current = open.top();
    open.pop();

    if (current.pathCost != distance[current.node]) continue;
    if (current.node == target) break;

    const auto [x, y] = fromNodeId(current.node, width);
    for (int direction = 0; direction < 4; ++direction) {
      const int nx = x + dx[direction];
      const int ny = y + dy[direction];
      if (nx < 0 || nx >= width || ny < 0 || ny >= height) continue;

      const int next = toNodeId(nx, ny, width);
      if (bannedNodes.count(next) || bannedEdges.count({current.node, next})) continue;
      const double terrainCost = grid[ny][nx];
      if (terrainCost < 0.0) continue;

      const double nextCost = current.pathCost + terrainCost;
      if (nextCost < distance[next]) {
        distance[next] = nextCost;
        parent[next] = current.node;
        open.push({nextCost + heuristic[next], nextCost, next});
      }
    }
  }

  if (!std::isfinite(distance[target])) return false;

  std::vector<int> reversed;
  for (int node = target; node != -1; node = parent[node]) {
    reversed.push_back(node);
    if (node == start) break;
  }
  if (reversed.empty() || reversed.back() != start) return false;

  std::reverse(reversed.begin(), reversed.end());
  result.nodes = std::move(reversed);
  result.cost = distance[target];
  return true;
}

double pathCost(const std::vector<int> &path,
                const std::vector<std::vector<double>> &grid, int width) {
  double total = 0.0;
  for (std::size_t i = 1; i < path.size(); ++i) {
    const auto [x, y] = fromNodeId(path[i], width);
    total += grid[y][x];
  }
  return total;
}

std::vector<Route> yenKShortestPaths(
    const std::vector<std::vector<double>> &grid, int width, int height,
    int start, int target, int wanted) {
  std::vector<Route> accepted;
  std::set<std::vector<int>> knownPaths;
  std::priority_queue<Route, std::vector<Route>, RouteGreater> candidates;
  const std::vector<double> heuristic =
      reverseDistancesToTarget(grid, width, height, target);

  Route first;
  if (!shortestPath(grid, width, height, start, target, {}, {}, heuristic,
                    first)) {
    return accepted;
  }
  accepted.push_back(first);
  knownPaths.insert(first.nodes);

  for (int routeIndex = 1; routeIndex < wanted; ++routeIndex) {
    const std::vector<int> &previous = accepted.back().nodes;

    for (std::size_t spurIndex = 0; spurIndex + 1 < previous.size(); ++spurIndex) {
      std::vector<int> root(previous.begin(), previous.begin() + spurIndex + 1);
      const int spurNode = root.back();

      std::set<int> bannedNodes;
      for (std::size_t i = 0; i + 1 < root.size(); ++i) {
        bannedNodes.insert(root[i]);
      }

      std::set<std::pair<int, int>> bannedEdges;
      for (const Route &route : accepted) {
        if (route.nodes.size() > spurIndex + 1 && hasPrefix(route.nodes, root)) {
          bannedEdges.insert({route.nodes[spurIndex], route.nodes[spurIndex + 1]});
        }
      }

      Route spur;
      if (!shortestPath(grid, width, height, spurNode, target, bannedNodes,
                        bannedEdges, heuristic, spur)) {
        continue;
      }

      std::vector<int> combined(root.begin(), root.end() - 1);
      combined.insert(combined.end(), spur.nodes.begin(), spur.nodes.end());
      if (knownPaths.insert(combined).second) {
        candidates.push({combined, pathCost(combined, grid, width)});
      }
    }

    if (candidates.empty()) break;
    accepted.push_back(candidates.top());
    candidates.pop();
  }

  return accepted;
}

json pathToJson(const std::vector<int> &path, int width) {
  json output = json::array();
  for (int node : path) {
    const auto [x, y] = fromNodeId(node, width);
    output.push_back({{"x", x}, {"y", y}});
  }
  return output;
}

void printError(const std::string &code, const std::string &message) {
  json output = {{"status", "error"}, {"error_code", code}, {"message", message}};
  std::cout << output.dump(2) << '\n';
}

}  // namespace

int main(int argc, char *argv[]) {
  if (argc > 2) {
    printError("USAGE", "استخدم: rover_pathfinder [مسار_ملف_JSON]، أو ضع input.json في مجلد التشغيل.");
    return 2;
  }

  const std::string inputPath = argc == 2 ? argv[1] : "input.json";
  std::ifstream inputFile(inputPath, std::ios::binary);
  if (!inputFile) {
    printError("FILE_NOT_FOUND", "تعذر فتح ملف الإدخال: " + inputPath);
    return 1;
  }

  try {
    json document;
    inputFile >> document;
    if (!document.is_object()) {
      throw std::runtime_error("يجب أن يكون محتوى الملف كائن JSON.");
    }

    if (!document.contains("width") || !document.contains("height")) {
      throw std::runtime_error("يجب توفير width و height.");
    }
    const int width = readInteger(document.at("width"), "width");
    const int height = readInteger(document.at("height"), "height");
    if (width <= 0 || height <= 0) {
      throw std::runtime_error("يجب أن يكون width و height أكبر من صفر.");
    }
    const std::int64_t cellCount = static_cast<std::int64_t>(width) * height;
    if (cellCount > kMaxGridCells) {
      throw std::runtime_error("حجم الشبكة أكبر من الحد المدعوم (مليون خلية).");
    }

    const Coordinate start = readCoordinate(document, "start");
    const Coordinate target = readCoordinate(document, "target");
    const auto inBounds = [width, height](const Coordinate &point) {
      return point.x >= 0 && point.x < width && point.y >= 0 && point.y < height;
    };
    if (!inBounds(start) || !inBounds(target)) {
      throw std::runtime_error("إحداثيات البداية أو الهدف خارج حدود الشبكة.");
    }

    if (!document.contains("grid") || !document.at("grid").is_array()) {
      throw std::runtime_error("grid يجب أن تكون مصفوفة ثنائية الأبعاد.");
    }
    const json &inputGrid = document.at("grid");
    if (inputGrid.size() != static_cast<std::size_t>(height)) {
      throw std::runtime_error("عدد صفوف grid لا يساوي height.");
    }

    std::vector<std::vector<double>> grid(
        static_cast<std::size_t>(height),
        std::vector<double>(static_cast<std::size_t>(width)));
    for (int y = 0; y < height; ++y) {
      if (!inputGrid.at(static_cast<std::size_t>(y)).is_array() ||
          inputGrid.at(static_cast<std::size_t>(y)).size() !=
              static_cast<std::size_t>(width)) {
        throw std::runtime_error("كل صف في grid يجب أن يحتوي width قيمة.");
      }
      for (int x = 0; x < width; ++x) {
        const json &cell = inputGrid.at(static_cast<std::size_t>(y)).at(
            static_cast<std::size_t>(x));
        if (!cell.is_number()) {
          throw std::runtime_error("كل خلية في grid يجب أن تكون رقمًا.");
        }
        const double value = cell.get<double>();
        if (!std::isfinite(value)) {
          throw std::runtime_error("قيم grid يجب أن تكون أرقامًا منتهية.");
        }
        grid[y][x] = value;
      }
    }

    if (grid[start.y][start.x] < 0.0) {
      throw std::runtime_error("خلية البداية عائق؛ يجب أن تكون تكلفتها صفرًا أو أكبر.");
    }
    if (grid[target.y][target.x] < 0.0) {
      throw std::runtime_error("خلية الهدف عائق؛ يجب أن تكون تكلفتها صفرًا أو أكبر.");
    }

    const int startId = toNodeId(start.x, start.y, width);
    const int targetId = toNodeId(target.x, target.y, width);
    const std::vector<Route> routes =
        yenKShortestPaths(grid, width, height, startId, targetId, kMaxPaths);

    if (routes.empty()) {
      json output = {
          {"status", "error"},
          {"error_code", "NO_PATH"},
          {"message", "لا يوجد مسار آمن بين البداية والهدف."},
          {"paths", json::array()}};
      std::cout << output.dump(2) << '\n';
      return 0;
    }

    json output = {
        {"status", "success"},
        {"message", "تم العثور على المسارات مرتبة حسب أقل تكلفة."},
        {"paths", json::array()}};
    for (std::size_t i = 0; i < routes.size(); ++i) {
      output["paths"].push_back({
          {"rank", i + 1},
          {"cost", routes[i].cost},
          {"path", pathToJson(routes[i].nodes, width)}});
    }
    // Compatibility with the previous response shape: path is the best route.
    output["path"] = pathToJson(routes.front().nodes, width);

    std::cout << output.dump(2) << '\n';
    return 0;
  } catch (const json::parse_error &) {
    printError("INVALID_JSON", "ملف الإدخال ليس JSON صالحًا.");
    return 1;
  } catch (const std::exception &error) {
    printError("INVALID_INPUT", error.what());
    return 1;
  }
}
